function farmWorldNormalize(raw){
 const d={schema:1,activeWorldId:ssRegion,position:{x:6.5,y:16.5},entryCount:0};
 if(!raw||typeof raw!=='object'||Array.isArray(raw))return d;
 if(raw.activeWorldId===FARM_WORLD_ID)d.activeWorldId=FARM_WORLD_ID;
 if(raw.position&&Number.isFinite(raw.position.x)&&Number.isFinite(raw.position.y)){
  const x=Math.max(2.5,Math.min(FARM_W-2.5,raw.position.x)),y=Math.max(2.5,Math.min(FARM_H-2.5,raw.position.y));
  if(!farmWorldBlocked(x,y))d.position={x,y};
 }
 d.entryCount=Number.isSafeInteger(raw.entryCount)?Math.min(100000,Math.max(0,raw.entryCount)):0;
 return d;
}
function farmWorldBlocked(x,y){
 if(!Number.isFinite(x)||!Number.isFinite(y)||x<1.4||y<1.4||x>FARM_W-2.4||y>FARM_H-2.4)return true;
 const rect=(x1,y1,x2,y2)=>x>x1&&x<x2&&y>y1&&y<y2;
 if(rect(8,4,15,10)||rect(33,4,41,11)||rect(34,22,43,30)||rect(4,22,8,27))return true;
 for(const [tx,ty] of [[12,24],[16,25],[40,16],[7,30],[43,12],[17,4]])if(Math.hypot(x-tx,y-ty)<.83)return true;
 return false;
}

function archWorldMovement(){
 if(m25bBusy)return {x:0,z:0};
 let x=(archKeys.has('d')||archKeys.has('arrowright')?1:0)-(archKeys.has('a')||archKeys.has('arrowleft')?1:0);
 let f=(archKeys.has('w')||archKeys.has('arrowup')?1:0)-(archKeys.has('s')||archKeys.has('arrowdown')?1:0);
 x+=archJoy.x;f-=archJoy.y;
 let length=Math.max(1,Math.hypot(x,f));x/=length;f/=length;
 // Screen-relative motion must use the exact same ground-plane basis as archCameraSetup:
 // camera.right=(cos(yaw), 0, -sin(yaw)), camera.forward=(-sin(yaw), 0, -cos(yaw)).
 // In M2.4, +f*sin(yaw) reversed forward X whenever the camera rotated.
 const cy=Math.cos(archYaw),sy=Math.sin(archYaw);
 return {x:x*cy-f*sy,z:-x*sy-f*cy};
}

function archPyramid(out,x,y,z,r,h,color){

function archCylinder(out,x,y,z,r,h,color,sides=9){

function archCameraSetup(){
 const target=[archPlayer.x,1.2,archPlayer.z],radius=14.7;
 const eye=[target[0]+Math.sin(archYaw)*Math.cos(archPitch)*radius,target[1]+Math.sin(archPitch)*radius+2,target[2]+Math.cos(archYaw)*Math.cos(archPitch)*radius];
 const norm=a=>{const d=Math.hypot(...a)||1;return a.map(v=>v/d)};
 const forward=norm(target.map((v,i)=>v-eye[i]));
 const right=norm([-forward[2],0,forward[0]]); // right = cross(forward, world-up)
 const up=[right[1]*forward[2]-right[2]*forward[1],right[2]*forward[0]-right[0]*forward[2],right[0]*forward[1]-right[1]*forward[0]];
 return {eye,forward,right,up};
}
function archProject(p,cam){
 const d=p.map((x,i)=>x-cam.eye[i]);
 const dot=v=>d[0]*v[0]+d[1]*v[1]+d[2]*v[2];
 const depth=dot(cam.forward);if(depth<=.55)return null;
 const focal=Math.min(archW*1.35,archH*1.85),sx=archW*.5+dot(cam.right)*focal/depth,sy=archH*.56-dot(cam.up)*focal/depth;
 return {x:sx,y:sy,depth};
}
function archLitColor(t,depth){
 const [a,b,c]=t.v,ax=b[0]-a[0],ay=b[1]-a[1],az=b[2]-a[2],bx=c[0]-a[0],by=c[1]-a[1],bz=c[2]-a[2];
 let nx=ay*bz-az*by,ny=az*bx-ax*bz,nz=ax*by-ay*bx,len=Math.hypot(nx,ny,nz)||1;
 const light=Math.max(.48,Math.min(1.1, .84+(.18*nx+.65*ny+.5*nz)/len*.28));
 const fog=Math.max(0,Math.min(.56,(depth-15)/40));
 return t.color.map((v,i)=>Math.max(0,Math.min(255,Math.round(v*light*(1-fog)+[14,29,39][i]*fog))));
}

function archShader(gl,type,src){let s=gl.createShader(type);gl.shaderSource(s,src);gl.compileShader(s);if(!gl.getShaderParameter(s,gl.COMPILE_STATUS))throw Error('shader:'+gl.getShaderInfoLog(s));return s;}
function archTryWebGL(){
 let gl;try{gl=archGpuCanvas.getContext('webgl',{antialias:true,alpha:false,depth:true,powerPreference:'low-power'});}catch(e){return false;}
 if(!gl)return false;
 try{
  let program=gl.createProgram();
  gl.attachShader(program,archShader(gl,gl.VERTEX_SHADER,'attribute vec3 a_position; attribute vec3 a_color; varying vec3 v_color; void main(){ gl_Position=vec4(a_position,1.0); v_color=a_color;}'));
  gl.attachShader(program,archShader(gl,gl.FRAGMENT_SHADER,'precision mediump float; varying vec3 v_color; void main(){gl_FragColor=vec4(v_color,1.0);}'));
  gl.linkProgram(program);if(!gl.getProgramParameter(program,gl.LINK_STATUS))throw Error('link:'+gl.getProgramInfoLog(program));
  archGl=gl;archProgram=program;archBuffer=gl.createBuffer();archGlPosition=gl.getAttribLocation(program,'a_position');archGlColor=gl.getAttribLocation(program,'a_color');
  gl.enable(gl.DEPTH_TEST);gl.depthFunc(gl.LEQUAL);gl.disable(gl.CULL_FACE);return true;
 }catch(err){console.warn('arch WebGL unavailable:',err);return false;}
}
