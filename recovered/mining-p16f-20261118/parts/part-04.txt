    state.bag.seed_carrot+=2;
   }
   q.completed++;q.readyAt=state.clock+LIVING_NPCS[id].cooldown;state.stats.quests++;award=LIVING_NPCS[id].payout;
   message=`${LIVING_NPCS[id].name} 의뢰 완료 · +${award} 크레딧`;break;}
  case 'home.build':{
   if(cmd.target!=='home')return bad('주택 부지가 아니에요');
   if(state.houseBuilt)return bad('이미 주택을 지었어요');
   if(stock('wood')<12||stock('stone')<6)return bad('원목 12개와 돌 6개가 필요해요');
   inv.wood-=12;inv.stone-=6;state.houseBuilt=true;message='첫 번째 정착 주택을 완성했어요';break;}
  default:return bad('지원하지 않는 행동이에요');
 }
 state.revision++;state.appliedIds.push(cmd.id);if(state.appliedIds.length>64)state.appliedIds.shift();
 state.events.push({id:cmd.id,kind,at:Math.floor(state.clock),actor:'local'});if(state.events.length>14)state.events.shift();
 return {ok:true,state,inv,coins:context.coins+award,message};
}

function ssVeinTypeV3(x,y,depth,biome){
 if(depth<6)return null;
 const cellW=18,cellH=14,cx=Math.floor(x/cellW),cy=Math.floor(y/cellH);
 let best=null,bestScore=Infinity;
 for(let ax=cx-1;ax<=cx+1;ax++)for(let ay=cy-1;ay<=cy+1;ay++){
  if(rand(ax,ay,1301)>.9)continue;
  const mx=(ax+.5)*cellW+(rand(ax,ay,1302)-.5)*5;
  const my=(ay+.5)*cellH+(rand(ax,ay,1303)-.5)*4;
  const theta=(rand(ax,ay,1304)-.5)*Math.PI*1.5;
  const cs=Math.cos(theta),sn=Math.sin(theta),dx=x+.5-mx,dy=y+.5-my;
  const u=dx*cs+dy*sn,v=-dx*sn+dy*cs;
  const halfLen=4.4+rand(ax,ay,1305)*4.8;
  const width=.96+rand(ax,ay,1306)*.74;
  const phase=rand(ax,ay,1307)*Math.PI*2;
  const bend=.9*Math.sin(u*.32+phase);
  const taper=.48+.52*Math.max(0,1-Math.abs(u)/halfLen);
  let score=Infinity;
  if(Math.abs(u)<=halfLen)score=Math.abs(v-bend)/(width*taper);
  const isBranch=rand(ax,ay,1308)<.28;
  if(isBranch){
   const branchBase=rand(ax,ay,1309)*halfLen*.7-halfLen*.35;
   const rel=u-branchBase;
   const branchSide=rand(ax,ay,1310)<.5?-1:1;
   if(rel>0&&rel<halfLen*.63){
    const branchV=bend+branchSide*rel*.72;
    score=Math.min(score,Math.abs(v-branchV)/(.56+.15*rand(ax,ay,1311)));
   }
  }
  if(score>1||score>=bestScore)continue;
  const r=rand(ax,ay,1312);
  let ore=r>.9&&biome==='forest'&&depth>9?'amber':
    r>.92&&(biome==='desert'||biome==='snow')&&depth>10?'opal':
    r>.86&&depth>22?'crystal':r>.59&&depth>12?'iron':r>.24?'copper':'orecoal';
  best=ore;bestScore=score;
 }
 return best;
}
