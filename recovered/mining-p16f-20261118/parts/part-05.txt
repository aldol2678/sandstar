 if((state.coins||0)<rule.cost)reasons.push(`크레딧 ${state.coins||0}/${rule.cost}C (부족 ${rule.cost-(state.coins||0)}C)`);
 for(const [id,qty] of rule.items){
  const available=forgeAvailable(id,state);if(available>=qty)continue;
  const reserved=forgeReserved(id,state),held=Math.max(0,Math.floor(Number(state.inv?.[id])||0));
  reasons.push(`${MATERIAL_NAMES[id]} ${available}/${qty}개 (부족 ${qty-available}개${reserved?`, 보유 ${held}개 중 수리용 ${reserved}개 예약`:''})`);
 }
 return reasons;
}
function forgeCanAttempt(state=game){
 const rule=forgeRule(state), forge=state.pickaxeForge;
 if(!rule||!forge||!Number.isSafeInteger(forge.attemptCounter)||forge.attemptCounter===Number.MAX_SAFE_INTEGER)return false;
 return Number.isSafeInteger(state.coins)&&state.coins>=rule.cost&&rule.items.every(([id,qty])=>forgeAvailable(id,state)>=qty);
}
function forgeRoll(state=game){
 const f=state.pickaxeForge;
 return hash(Math.imul(state.seed|0,1597334677)^Math.imul((f.attemptCounter+1)|0,1664525)^Math.imul((f.rank+1)|0,1103515245)^0x6a09e667);
}
function attemptPickaxeForge(){
 if(performance.now()<forgeTapLockUntil)return false;
 if(!forgeRule())return false;
 if(!forgeCanAttempt()){flash('강화 조건 미충족 · '+forgeShortageReasons().join(' / '));return false;}
 let outcome=null;
 const ok=safeMarketTransaction(draft=>{
  if(!forgeCanAttempt(draft))return false;
  const f=draft.pickaxeForge, rule=forgeRule(draft), catalyst=ssP14EffectiveCatalyst(draft);
  const chance=Math.min(100,forgeChance(draft)+SS_P14_CATALYSTS[catalyst].bonus);
  const success=forgeRoll(draft)*100<chance;
  if(catalyst!=='none'){if(draft.treasureP13.materials[catalyst]<1)return false;draft.treasureP13.materials[catalyst]-=1;}
  const nextRank=f.rank+1;
  if(success){
   draft.coins-=rule.cost;
   for(const [id,qty] of rule.items)draft.inv[id]-=qty;
   f.rank=nextRank;
   f.failStreak=0;
  }else{
   draft.coins-=Math.ceil(rule.cost/2);
   f.failStreak=Math.min(4,f.failStreak+1);
  }
  f.attemptCounter++;
  draft.economy.balanceMetrics.forgeAttempts++;
  draft.economy.balanceMetrics.forgeCreditsSpent+=success?rule.cost:Math.ceil(rule.cost/2);
  if(!success)draft.economy.balanceMetrics.forgeFailures++;
  outcome={success,stage:nextRank,charged:success?rule.cost:Math.ceil(rule.cost/2),catalyst};
  return true;
 });
 if(!ok)return false;
 forgeTapLockUntil=performance.now()+450;
 ssP14Catalyst='none';
 ssForgeLastOutcome={...outcome,rank:game.pickaxeForge.rank,coins:game.coins,nextChance:forgeChance()};
 forgeMenu();
 if(outcome.success){
  flash(`곡괭이 +${outcome.stage} 강화 성공! 채굴력 +${Math.round(forgeRankBonus(game.pickaxeForge.rank)*100)}%`);
  playSound('pickup_ore');
 }else{
  flash(`강화 실패 · 단계 유지 · ${outcome.charged} 크레딧 소모\n다음 성공률 ${forgeChance()}%`);
  playSound('stone_hit');
 }
 return true;
}

function sellOre(id,quantity){
 const entry=ORE_SALE_ITEMS.find(item=>item[0]===id);
 if(!entry)return false;
 const ok=safeMarketTransaction(draft=>{
  const available=sellableOreCount(id,draft);
  const amount=quantity==='all'?available:Math.min(available,quantity);
  if(!Number.isSafeInteger(amount)||amount<=0)return false;
  const value=amount*entry[2];
  if(!Number.isSafeInteger(value)||!Number.isSafeInteger(draft.coins+value))return false;
  draft.inv[id]-=amount;draft.coins+=value;
  draft.economy.totalOreSold+=amount;draft.economy.totalCreditsEarned+=value;draft.economy.balanceMetrics.oreSold+=amount;draft.economy.balanceMetrics.creditsSold+=value;
  return true;
 });
 if(ok){market();flash(`${entry[1]} ${quantity==='all'?'전량':quantity+'개'} 판매 완료`);}else if(sellableOreCount(id)===0){flash('판매 가능한 광물이 없어요. 보존 재료를 확인하세요.');}
 return ok;
}
function sellAllOres(){
 let totalCount=0,totalCredit=0;
 const ok=safeMarketTransaction(draft=>{
  for(const [id,label,price] of ORE_SALE_ITEMS){
   const n=sellableOreCount(id,draft);if(n===0)continue;
   draft.inv[id]-=n;draft.coins+=n*price;
   totalCount+=n;totalCredit+=n*price;
  }
  if(totalCount<=0||!Number.isSafeInteger(draft.coins))return false;
  draft.economy.totalOreSold+=totalCount;draft.economy.totalCreditsEarned+=totalCredit;draft.economy.balanceMetrics.oreSold+=totalCount;draft.economy.balanceMetrics.creditsSold+=totalCredit;
  return Number.isSafeInteger(draft.economy.totalCreditsEarned);
 });
 if(ok){market();flash(`광물 ${totalCount}개 판매: +${totalCredit} 크레딧`);}else if(totalCount===0){flash('판매 가능한 광물이 없어요.');}
 return ok;
}
