function p16HammerEligible(x,y){

    x<0||x>=WW||y<0||y>=WH-5||y<terrain(x)+3||stationGeneratedTile(x,y)!==undefined)return false;

function p16HammerWave(x,y){
 if(game.tool!=='hammer'||ssRegion!=='mine'||paused)return false;
 const rule=P16_HAMMER_CORES[p16HammerRank()];
 let broken=0,affected=0;const seen=new Set([key(x,y)]),queue=[{x,y,depth:0}];
 // Finite radius, 24 cells inspected per impact, no global flood fill.
 while(queue.length&&affected<24){
  const at=queue.shift();if(at.depth>rule.maxDepth)continue;
  for(const [dx,dy] of P16_HAMMER_DIRS){
   const nx=at.x+dx,ny=at.y+dy,k=key(nx,ny);
   if(seen.has(k))continue;seen.add(k);
   if(!p16HammerEligible(nx,ny))continue;
   affected++;
   const t=tile(nx,ny),hp=tileHP(t),last=p16HammerFractures.get(k);
   // Depth attenuation prevents a single strike from clearing a large cavern.
   const damage=hp*rule.fracture*(at.depth===0?1:.68);
   const next=(last?.type===t?last.damage:0)+damage;
   if(next+1e-6>=hp&&broken<rule.chainMax){
    p16HammerFractures.delete(k);
    if(breakBlock(nx,ny,t,true)){
     broken++;p16HammerStats.chainBroken++;
     if(at.depth<rule.maxDepth)queue.push({x:nx,y:ny,depth:at.depth+1});
    }
   }else{
    p16HammerFractures.set(k,{x:nx,y:ny,type:t,damage:Math.min(hp-.001,next)});
    p16HammerStats.cracked++;
    mineSpark(nx,ny,t,false);
   }
   if(p16HammerFractures.size>128)p16HammerFractures.delete(p16HammerFractures.keys().next().value);
   if(broken>=rule.chainMax||affected>=24)break;
  }
 }
 p16HammerStats.waves++;p16HammerStats.last={x,y,rank:p16HammerRank(),affected,broken};
 return true;
}

function p16DFindNextOre(x,y,type){
 const candidates=[];
 for(let dy=-2;dy<=2;dy++)for(let dx=-2;dx<=2;dx++){
  if(dx===0&&dy===0||Math.hypot(dx,dy)>2.25)continue;
  const tx=x+dx,ty=y+dy;
  if(tx<0||tx>=WW||ty<0||ty>=WH-5||stationGeneratedTile(tx,ty)!==undefined)continue;
  if(Object.prototype.hasOwnProperty.call(game.edits,key(tx,ty)))continue;
  if(tile(tx,ty)!==type)continue;
  candidates.push({x:tx,y:ty,d:Math.hypot(dx,dy),sort:(ty*WW+tx)});
 }
 candidates.sort((a,b)=>a.d-b.d||a.sort-b.sort);
 return candidates.length?{x:candidates[0].x,y:candidates[0].y}:null;
}
function p16DOnMinedBlock(x,y,type,natural){

 // The target was just removed from the world, so do not revalidate tile() here.
 const followed=!!old&&now<p16DTrace.expires&&old.x===x&&old.y===y&&type===p16DTrace.ore;
 const next=p16DFindNextOre(x,y,type);
 if(!next){p16DReset();return;}
 p16DTrace.chain=followed?Math.min(3,p16DTrace.chain+1):1;
 p16DTrace.ore=type;p16DTrace.hint=next;p16DTrace.expires=now+P16D_TRACE_MS;

}

// A shaft may have a clear exit but no suitable solid anchor three tiles above.
// Use a real, unprotected bank tile beside the lip; every player move is collision-checked.
function ropeSurfaceExitPlan(){
 if(!p||paused)return null;
 const column=Math.floor(p.x),surfaceY=terrain(column),depth=p.y-surfaceY;
 if(depth<-.5||depth>6.65)return null;
 const directions=input.left?[-1,1]:[1,-1];
 for(const side of directions){
  const landingX=column+.5+side*1.08;
  // Actual top bank includes the generated grass tile above terrain().
  let groundY=null;
  for(let y=Math.max(1,terrain(landingX)-4);y<=terrain(landingX)+2;y++){
   if(solid(tile(Math.floor(landingX),y))){groundY=y;break;}
  }
  if(groundY===null||Math.abs(groundY-surfaceY)>2.5)continue;
  const bankTile=tile(Math.floor(landingX),groundY);
  if(!solid(bankTile)||bankTile==='stationrock'||bankTile==='bedrock')continue;
  const exitY=Math.min(surfaceY-1.35,groundY-.35);
  if(!Number.isFinite(exitY)||exitY<1||collides(landingX,exitY))continue;
  // A narrow shaft must be genuinely open all the way up. No phase-through mining.
  let clear=true;
  for(let y=p.y;y>=exitY;y-=.10){if(collides(p.x,y)){clear=false;break;}}
  if(!clear||collides(p.x,exitY))continue;
  const n=Math.ceil(Math.abs(landingX-p.x)/.1);
  for(let i=1;i<=n;i++)if(collides(p.x+(landingX-p.x)*i/n,exitY)){clear=false;break;}
  if(!clear)continue;
  return {surfaceY,exitY,landingX,groundY,anchor:{x:Math.floor(landingX),y:groundY,
   faceX:column+(side>0?1:0),faceY:groundY}};
 }
