const ANIM_FALLBACK='idle';
let animName='idle',animClock=0,landClock=0,animWasGrounded=null;
function resetSpriteAnimation(){animName='idle';animClock=0;landClock=0;animWasGrounded=null;}
function updateSpriteAnimation(dt){
 if(!p||paused)return;
 const grounded=Boolean(p.grounded);
 if(animWasGrounded===false&&grounded&&!rope.active)landClock=.235;
 animWasGrounded=grounded;
 if(!grounded)landClock=0;
 else landClock=Math.max(0,landClock-dt);
 let desired='idle';
 if(rope.active)desired='rope_climb';
 else if(!grounded&&airLeapControl&&p.vy<3.5)desired='rope_leap';
 else if(!grounded&&p.vy<-.45)desired='jump';
 else if(!grounded&&p.vy>.45)desired='fall';
 else if(landClock>0)desired='land';
 else if(Math.abs(p.vx)>.48)desired='walk';
 if(desired!==animName){animName=desired;animClock=0;}
 else animClock+=dt;
}
function currentMinerFrame(){
 const clip=MINER_SPRITE_INFO.animations[animName]||MINER_SPRITE_INFO.animations[ANIM_FALLBACK];
 const frames=clip.frames;
 const duration=Math.max(.02,clip.frame_duration_ms/1000);
 const position=rope.active&&rope.attached&&animName==='rope_climb'?0:Math.floor(animClock/duration);
 return frames[clip.loop?position%frames.length:Math.min(frames.length-1,position)];
}
function currentMiningDirection(){
 if(stickActive&&stickDir)return [stickDir.dx,stickDir.dy];
 if(holdTarget){let dx=holdTarget.x+.5-p.x,dy=holdTarget.y+.5-(p.y-PLAYER_CENTER_FROM_FEET);return [Math.sign(dx),Math.sign(dy)];}
 return [p.facing||1,0];
}
function isMiningAnimationActive(){return (stickActive||Boolean(holdTarget)||input.act)&&Boolean(miningTarget())&&!rope.active;}
function minerVisualFacing(){
 if(!isMiningAnimationActive())return p.facing||1;
 const dx=currentMiningDirection()[0];return dx===0?(p.facing||1):Math.sign(dx);
}
function drawEquipmentOverlay(x,foot){
 if(!isMiningAnimationActive() || (!minerToolsReady && !forgeAppearanceReady))return;
 const toolId=Object.prototype.hasOwnProperty.call(MINER_TOOL_INFO.tools,game.tool)?game.tool:'pickaxe';
 const tool=MINER_TOOL_INFO.tools[toolId];
 const index=Math.floor(performance.now()/tool.frame_duration_ms)%tool.frames;
 const vec=currentMiningDirection();
 const dx=vec[0],dy=vec[1];
 const facing=minerVisualFacing();
 const a=Math.atan2(dy,Math.abs(dx)||.05);
 // Mirror the tool toward the target side, then rotate toward the selected upward/downward axis.
 ctx.save();
 ctx.translate(x+(dx||facing)*T*.27,foot-T*.58+dy*T*.13);
 ctx.rotate(a*facing);
 ctx.scale(facing,1);
 ctx.imageSmoothingEnabled=false;
 const useForgeArt=toolId==='pickaxe'&&forgeAppearanceReady;
 const toolAtlas=useForgeArt?forgeAppearanceImage:minerToolImage;
 const toolRow=useForgeArt?forgeAppearanceTier(game.pickaxeForge?.rank||0):tool.row;
 if((useForgeArt||minerToolsReady)) ctx.drawImage(toolAtlas,index*32,toolRow*40,32,40,-T*.4,-T*.64,T*.8,T);
 ctx.restore();
}
function drawPlayer(){
 if(!minerSpriteReady){
  const px=Math.round(p.x*T-camera.x),feet=Math.round(p.y*T-camera.y);
  ctx.save();ctx.translate(px,feet);ctx.scale(PLAYER_DRAW_SIZE/42,PLAYER_DRAW_SIZE/42);ctx.translate(-px,-feet);
  drawPlayerLegacy();ctx.restore();return;
 }
 const x=Math.round(p.x*T-camera.x), foot=Math.round(p.y*T-camera.y);
 const index=currentMinerFrame(), fw=MINER_SPRITE_INFO.frame_width, fh=MINER_SPRITE_INFO.frame_height, cols=MINER_SPRITE_INFO.columns;
 // A sprite frame's pivot is centered on the player collision footprint.
 ctx.save();
 ctx.translate(x,foot);
 ctx.scale(minerVisualFacing()<0?-1:1,1);
 ctx.imageSmoothingEnabled=false;
 // Grounded, dark single-pixel contour improves legibility on pale and dark rock alike.
 // It does not scale the sprite beyond its one-tile footprint.
 ctx.shadowColor='#091d28';ctx.shadowBlur=0;ctx.shadowOffsetX=1;ctx.shadowOffsetY=1;
 ctx.drawImage(minerSpriteImage,(index%cols)*fw,Math.floor(index/cols)*fh,fw,fh,-PLAYER_DRAW_SIZE/2,-PLAYER_DRAW_SIZE,PLAYER_DRAW_SIZE,PLAYER_DRAW_SIZE);
 ctx.shadowOffsetX=0;ctx.shadowOffsetY=0;
 ctx.restore();
 // Stowed pickaxe remains visible on the right side when not actively mining.
 // Purely visual: no collision, movement, reach or miningRate changes.
 if(game.tool==='pickaxe'&&forgeAppearanceReady&&!isMiningAnimationActive()){
  const tier=forgeAppearanceTier(game.pickaxeForge?.rank||0);
  ctx.save();ctx.translate(x,foot);ctx.scale(minerVisualFacing()<0?-1:1,1);ctx.imageSmoothingEnabled=false;
  ctx.drawImage(forgeAppearanceImage,0,tier*40,32,40,-5,-35,30,37);ctx.restore();
 }
 drawEquipmentOverlay(x,foot);
 if(isMiningAnimationActive()){
  const dir=currentMiningDirection(),a=performance.now()/110;
  const fx=Math.round(x+dir[0]*T*.43),fy=Math.round(foot-T*.55+dir[1]*T*.36);
  ctx.save();ctx.globalAlpha=.65+.25*(.5+.5*Math.sin(a));ctx.fillStyle='#fff0a8';
  ctx.fillRect(fx-1,fy-3,2,6);ctx.fillRect(fx-3,fy-1,6,2);ctx.restore();
 }
}

// 2.15-C: scan glows ONLY existing ore tiles within radius; never increases mining reach.
function drawAffinityRadar(){if(ssRegion==='life_forest_01')return;
 const radius=pickaxeScanRadius();if(!radius)return;
 const px=p.x,py=p.y-PLAYER_CENTER_FROM_FEET;
 const phase=.55+.30*Math.sin(performance.now()/390);
 ctx.save();ctx.strokeStyle='#d3b9ff';ctx.fillStyle='#f1deff';ctx.lineWidth=1.4;
 for(let dy=-radius;dy<=radius;dy++)for(let dx=-radius;dx<=radius;dx++){
  const x=Math.floor(px)+dx,y=Math.floor(py)+dy;
  if(Math.hypot(x+.5-px,y+.5-py)>radius+1e-9)continue;
  if(!visionVisible(x,y)||!ORES.has(tile(x,y)))continue;
  const sx=Math.round(x*T-camera.x),sy=Math.round(y*T-camera.y);
  if(sx<-T||sy<-T||sx>viewW||sy>viewH)continue;
  ctx.globalAlpha=phase;ctx.strokeRect(sx+2,sy+2,T-4,T-4);
  ctx.fillRect(sx+T/2-1,sy+3,2,5);ctx.fillRect(sx+T/2-3,sy+5,6,2);
 }
 ctx.restore();
}
